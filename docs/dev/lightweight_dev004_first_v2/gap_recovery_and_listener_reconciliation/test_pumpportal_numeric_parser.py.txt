def encoded(byte, size=32):
    alphabet = '123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz'
    value = int.from_bytes(bytes([byte])*size, 'big')
    out = ''
    while value:
        value, index = divmod(value, 58)
        out = alphabet[index]+out
    return out


M, C, S = encoded(109), encoded(99), encoded(115, 64)

def baseline():

def extract(text, scope):
    tree = ast.parse(text)
    names = {NAME, '_compact_pumpportal_birth_retry', '_decode_compact_pumpportal_birth_retry'}
    nodes = [n for n in ast.walk(tree) if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)) and n.name in names]
    exec(compile(ast.Module(body=nodes, type_ignores=[]), str(SOURCE), 'exec'), scope)
    return scope[NAME]


class Target:
    def __init__(self, name, rows):
        self.name, self.rows = name, rows
    def exists(self):
        return self.name in self.rows
    def with_suffix(self, suffix):
        return Target(self.name+suffix, self.rows)


@pytest.fixture

    # Use actual writer admission/outcome methods, replacing only physical storage.

            core.setdefault(mint, (creator, creator, kwargs['create_tx_signature']))

        async def outcome(ident, mint, committed):

            def __enter__(self): return self

@pytest.mark.parametrize('data', VALID)
def test_valid_receipt_bytes_and_canonical_parity(harness, data):
    old, new = harness(baseline()), harness()
    old.produce(data); new.produce(data)
    assert old.snapshot() == new.snapshot()
    assert len(new.rows) == 1 and len(new.attempts) == 1
    row = next(iter(new.rows.values()))
    assert row['persistence_outcome'] == 'COMMITTED'
    assert row['validation_result'] == 'IDENTITIES_VALID_UNVERIFIED_CHAIN'


@pytest.mark.parametrize('data,reserve,cap', MALFORMED)
def test_malformed_metadata_no_longer_blocks_receipt_or_canonical(harness, data, reserve, cap):
    old = harness(baseline())
    with pytest.raises((TypeError, ValueError, OverflowError)):
        old.produce(data)
    assert not old.rows and not old.attempts
    new = harness(); new.produce(data)
    assert len(new.rows) == 1 and new.core[M] == (C,C,S)
    assert next(iter(new.rows.values()))['persistence_outcome'] == 'COMMITTED'
    assert new.actor._portal_vsol[M]['v_sol'] == reserve
    assert new.actor._portal_vsol[M]['mc_sol'] == cap


def test_duplicate_parity(harness):
    old, new = harness(baseline()), harness()
    for actor in [old,new]: actor.produce(); actor.produce()
    assert old.snapshot() == new.snapshot()
    assert len(new.rows) == len(new.core) == 1


def test_conflict_preserves_canonical_and_receipt_identity(harness):
    old, new = harness(baseline(), conflict=True), harness(conflict=True)
    old.produce(); new.produce({'marketCapSol':'bad'})
    assert old.snapshot() == new.snapshot()
    r = next(iter(new.rows.values()))
    assert r['persistence_outcome'] == 'COMMITTED_CONFLICT' and r['canonical_conflict']
    assert r['supplied_creator'] == C and new.core[M][0] != C


def test_receipt_failure_remains_fail_open_for_canonical(harness, caplog):
    new = harness(receipt_failure=True); new.produce({'marketCapSol':{ 'bad':1 }})
    assert not new.rows and new.core[M] == (C,C,S)
    assert 'stage=birth_provenance reason=FS_NO_SPACE' in caplog.text


def test_retry_payload_and_outcome_parity(harness):
    old, new = harness(baseline(), persistence_failure=True), harness(persistence_failure=True)
    old.produce(); new.produce({'marketCapSol':[1]})
    assert old.retry == new.retry and len(new.retry) == 1
    assert not new.core
    assert next(iter(new.rows.values()))['persistence_outcome'] == 'RETRY_PENDING'
    assert json.loads(new.retry[S])['mint'] == M


def test_only_requested_handler_changed():
    old, new = ast.parse(baseline()), ast.parse(SOURCE.read_text())
    def replace(tree):
        for n in ast.walk(tree):
            if isinstance(n, ast.AsyncFunctionDef) and n.name == NAME:
                n.body = [ast.Pass()]
        return ast.dump(tree, include_attributes=False)
    assert replace(old) == replace(new)
